from pathlib import Path


def replace(path: str, old: str, new: str) -> None:
    p = Path(path)
    text = p.read_text()
    if old not in text:
        raise SystemExit(f'Expected snippet not found in {path}: {old[:160]!r}')
    p.write_text(text.replace(old, new, 1))


def write(path: str, content: str) -> None:
    p = Path(path)
    p.parent.mkdir(parents=True, exist_ok=True)
    p.write_text(content)

write('packages/domain/src/expense.ts', r'''import { DomainInvariantError } from './errors';
import type { BusinessDayId, ExpenseId, ShopId, WorkerId } from './ids';
import type {
  Expense,
  ExpensePaidFrom,
  ManualExpense,
  ManualExpenseLifecycleSnapshot,
} from './models';
import { addMoney, ZERO_MONEY, type MoneyMinor } from './money';
import type { Instant } from './time';

export interface ManualExpenseValues {
  readonly description: string;
  readonly amountMinor: MoneyMinor;
  readonly paidFrom: ExpensePaidFrom;
  readonly note: string | null;
}

export interface ExpenseTotals {
  readonly totalExpensesMinor: MoneyMinor;
  readonly cashExpensesMinor: MoneyMinor;
}

export function manualExpenseLifecycle(expense: ManualExpense): ManualExpenseLifecycleSnapshot {
  return (
    expense.lifecycle ?? {
      revision: 0,
      updatedAt: null,
      updatedByWorkerId: null,
      deletedAt: null,
      deletedByWorkerId: null,
    }
  );
}

export function isExpenseDeleted(expense: Expense): boolean {
  return expense.kind === 'MANUAL' && manualExpenseLifecycle(expense).deletedAt !== null;
}

function nextRevision(expense: ManualExpense): number {
  const next = manualExpenseLifecycle(expense).revision + 1;
  if (!Number.isSafeInteger(next) || next <= 0) {
    throw new DomainInvariantError('Expense revision overflowed its safe range.');
  }
  return next;
}

export function normalizeManualExpenseValues(values: ManualExpenseValues): ManualExpenseValues {
  const description = values.description.trim();
  if (description.length === 0) {
    throw new DomainInvariantError('Expense description is required.');
  }
  if (description.length > 160) {
    throw new DomainInvariantError('Expense description cannot exceed 160 characters.');
  }
  if (values.amountMinor <= ZERO_MONEY) {
    throw new DomainInvariantError('Expense amount must be greater than zero.');
  }
  if (values.paidFrom !== 'CASH' && values.paidFrom !== 'OTHER') {
    throw new DomainInvariantError('Expense Paid From must be Cash or Other.');
  }
  const trimmedNote = values.note?.trim() ?? '';
  if (trimmedNote.length > 500) {
    throw new DomainInvariantError('Expense note cannot exceed 500 characters.');
  }
  return {
    description,
    amountMinor: values.amountMinor,
    paidFrom: values.paidFrom,
    note: trimmedNote.length === 0 ? null : trimmedNote,
  };
}

export function createManualExpense(
  identity: {
    readonly id: ExpenseId;
    readonly shopId: ShopId;
    readonly businessDayId: BusinessDayId;
    readonly createdByWorkerId: WorkerId;
    readonly createdAt: Instant;
  },
  values: ManualExpenseValues,
): ManualExpense {
  const normalized = normalizeManualExpenseValues(values);
  return {
    ...identity,
    kind: 'MANUAL',
    ...normalized,
    orderId: null,
    lifecycle: {
      revision: 0,
      updatedAt: null,
      updatedByWorkerId: null,
      deletedAt: null,
      deletedByWorkerId: null,
    },
  };
}

export function editManualExpense(
  expense: Expense,
  values: ManualExpenseValues,
  updatedAt: Instant,
  updatedByWorkerId: WorkerId,
): ManualExpense {
  if (expense.kind !== 'MANUAL') {
    throw new DomainInvariantError('System Delivery Failed expenses are locked.');
  }
  const lifecycle = manualExpenseLifecycle(expense);
  if (lifecycle.deletedAt !== null) {
    throw new DomainInvariantError('Deleted expenses cannot be edited.');
  }
  const normalized = normalizeManualExpenseValues(values);
  return {
    ...expense,
    ...normalized,
    lifecycle: {
      ...lifecycle,
      revision: nextRevision(expense),
      updatedAt,
      updatedByWorkerId,
    },
  };
}

export function deleteManualExpense(
  expense: Expense,
  deletedAt: Instant,
  deletedByWorkerId: WorkerId,
): ManualExpense {
  if (expense.kind !== 'MANUAL') {
    throw new DomainInvariantError('System Delivery Failed expenses are locked.');
  }
  const lifecycle = manualExpenseLifecycle(expense);
  if (lifecycle.deletedAt !== null) {
    throw new DomainInvariantError('Expense is already deleted.');
  }
  return {
    ...expense,
    lifecycle: {
      ...lifecycle,
      revision: nextRevision(expense),
      deletedAt,
      deletedByWorkerId,
    },
  };
}

export function calculateExpenseTotals(expenses: readonly Expense[]): ExpenseTotals {
  let totalExpensesMinor = ZERO_MONEY;
  let cashExpensesMinor = ZERO_MONEY;
  for (const expense of expenses) {
    if (expense.kind !== 'MANUAL' || isExpenseDeleted(expense)) continue;
    totalExpensesMinor = addMoney(totalExpensesMinor, expense.amountMinor);
    if (expense.paidFrom === 'CASH') {
      cashExpensesMinor = addMoney(cashExpensesMinor, expense.amountMinor);
    }
  }
  return { totalExpensesMinor, cashExpensesMinor };
}
''')

  createManualExpense,
  deleteManualExpense,

import type {

const DAY_ID = parseEntityId<BusinessDayId>('20000000-0000-4000-8000-000000000001');
const WORKER_ID = parseEntityId<WorkerId>('30000000-0000-4000-8000-000000000001');
const OTHER_WORKER_ID = parseEntityId<WorkerId>('30000000-0000-4000-8000-000000000002');

function manual(id: string, amount: number, paidFrom: 'CASH' | 'OTHER') {

    {
      id: parseEntityId<ExpenseId>(id),
      shopId: SHOP_ID,
      businessDayId: DAY_ID,
      createdByWorkerId: WORKER_ID,
      createdAt: instant('2026-08-18T10:00:00.000Z'),
    },
    { description: ' Taxi ', amountMinor: moneyMinor(amount), paidFrom, note: ' shift ride ' },
  );
}

describe('expense domain', () => {
  it('normalizes manual expense values and tracks audited edit/delete lifecycle', () => {
    const created = manual('40000000-0000-4000-8000-000000000001', 15_000, 'CASH');
    expect(created.description).toBe('Taxi');
    expect(created.note).toBe('shift ride');
    expect(manualExpenseLifecycle(created).revision).toBe(0);

    const edited = editManualExpense(
      created,
      { description: 'Fuel', amountMinor: moneyMinor(20_000), paidFrom: 'OTHER', note: '  ' },
      instant('2026-08-18T11:00:00.000Z'),
      OTHER_WORKER_ID,

    expect(edited.amountMinor).toBe(moneyMinor(20_000));
    expect(manualExpenseLifecycle(edited).revision).toBe(1);
    expect(manualExpenseLifecycle(edited).updatedByWorkerId).toBe(OTHER_WORKER_ID);

    const deleted = deleteManualExpense(
      edited,
      instant('2026-08-18T12:00:00.000Z'),
      WORKER_ID,

    expect(manualExpenseLifecycle(deleted).revision).toBe(2);

      shopId: SHOP_ID,
      businessDayId: DAY_ID,

      paidFrom: null,

      createdByWorkerId: WORKER_ID,
      createdAt: instant('2026-08-18T12:30:00.000Z'),
    };

    expect(calculateExpenseTotals([cash, other, deleted, deliveryFailed])).toEqual({
      totalExpensesMinor: moneyMinor(35_000),

      shopId: SHOP_ID,
      businessDayId: DAY_ID,

    expect(() =>

        {
          id: parseEntityId<ExpenseId>('40000000-0000-4000-8000-000000000007'),
          shopId: SHOP_ID,
          businessDayId: DAY_ID,
          createdByWorkerId: WORKER_ID,
          createdAt: instant('2026-08-18T13:00:00.000Z'),
        },

      ),
    ).toThrow(/greater than zero/);
  });
});

      async getById(id: ExpenseId) {
        return recordOrNull<Expense>(store('expenses').get(id));
      },
      async listByBusinessDay(businessDayId: BusinessDayId) {
        const all = (await requestResult(
          store('expenses').index('businessDayId').getAll(businessDayId),
        )) as Expense[];
        return all.sort(
          (left, right) =>
            right.createdAt.localeCompare(left.createdAt) || right.id.localeCompare(left.id),
        );
      },
      async put(expense: Expense) {
        const expenses = store('expenses');
        const existing = await recordOrNull<Expense>(expenses.get(expense.id));
        if (existing !== null && existing.kind !== expense.kind) {
          throw new Error(`Expense ${expense.id} cannot change kind.`);
        }
        if (existing?.kind === 'DELIVERY_FAILED') {
          throw new Error(`Delivery Failed expense ${expense.id} is locked.`);
        }
        if (expense.kind === 'DELIVERY_FAILED' && expense.orderId !== null) {
          const all = (await requestResult(expenses.getAll())) as Expense[];
          if (
            all.some(
              (candidate) =>
                candidate.kind === 'DELIVERY_FAILED' &&
                candidate.orderId === expense.orderId &&
                candidate.id !== expense.id,
            )
          ) {
            throw new Error(`Delivery Failed expense already exists for order ${expense.orderId}.`);
          }
