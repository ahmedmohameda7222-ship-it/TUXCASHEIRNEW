# Application Expenses service.
write('packages/application/src/expenses.ts', r'''import {
  DomainInvariantError,
  calculateExpenseTotals,
  createManualExpense,
  deleteManualExpense,
  editManualExpense,
  isExpenseDeleted,
  manualExpenseLifecycle,
  parseEntityId,
  type AuditEvent,
  type AuditEventId,
  type BusinessDayId,
  type EntityId,
  type Expense,
  type ExpenseId,
  type ExpensePaidFrom,
  type Instant,
  type JsonValue,
  type MoneyMinor,
  type OutboxEvent,
  type OutboxEventId,
  type ShopId,
  type Worker,
} from '@tux/domain';
import type {
  OperationsDatabase,
  OperationsTransaction,
  OperatorSessionReadModel,
} from '@tux/persistence';
import { ApplicationCommandCoordinator } from './commandCoordinator';
import type { ApplicationError } from './errors';
import { err, ok, type Result } from './result';

export interface ExpensesRuntime {
  now(): Instant;
  createUuid(): string;
}

export interface ManualExpenseInput {
  readonly description: string;
  readonly amountMinor: MoneyMinor;
  readonly paidFrom: ExpensePaidFrom;
  readonly note: string | null;
}

export interface EditManualExpenseInput extends ManualExpenseInput {
  readonly expenseId: ExpenseId;
}

export interface ExpensesLedger {
  readonly shopId: ShopId;
  readonly businessDayId: BusinessDayId;
  readonly loadedAt: Instant;
  readonly expenses: readonly Expense[];
  readonly totalExpensesMinor: MoneyMinor;
  readonly cashExpensesMinor: MoneyMinor;
}

export type ExpensesLedgerResult = Result<ExpensesLedger, ApplicationError>;
export type ExpenseMutationResult = Result<Expense, ApplicationError>;

interface MutationContext {
  readonly shopId: ShopId;
  readonly businessDayId: BusinessDayId;
  readonly operator: Worker;
}

function persistenceError(message: string, cause: unknown): ApplicationError {
  return { code: 'LOCAL_PERSISTENCE_ERROR', message, cause };
}

function mutationError(cause: unknown): ApplicationError {
  if (cause instanceof DomainInvariantError) {
    return { code: 'CONFLICT_ERROR', message: cause.message, cause };
  }
  return persistenceError('The expense change could not be committed locally.', cause);
}

export class OperationsExpensesService {
  readonly #database: OperationsDatabase;
  readonly #readModel: OperatorSessionReadModel;
  readonly #runtime: ExpensesRuntime;
  readonly #coordinator: ApplicationCommandCoordinator;

  constructor(
    database: OperationsDatabase,
    readModel: OperatorSessionReadModel,
    runtime: ExpensesRuntime,
    coordinator = new ApplicationCommandCoordinator(),
  ) {
    this.#database = database;
    this.#readModel = readModel;
    this.#runtime = runtime;
    this.#coordinator = coordinator;
  }

  async loadLedger(): Promise<ExpensesLedgerResult> {
    return this.#coordinator.runExclusive(async () => {
      try {
        const shops = await this.#readModel.listActiveShops();
        const shop = shops.length === 1 ? shops[0] : undefined;
        if (shop === undefined) {
          return err({
            code: 'CONFLICT_ERROR',
            message: 'This device is not assigned to exactly one active shop.',
          });
        }
        const ledger = await this.#database.transaction(async (transaction) => {
          const day = await transaction.businessDays.getOpenForShop(shop.id);
          if (day === null || day.status !== 'OPEN') return null;
          const all = await transaction.expenses.listByBusinessDay(day.id);
          const expenses = all
            .filter((expense) => !isExpenseDeleted(expense))
            .sort(
              (left, right) =>
                right.createdAt.localeCompare(left.createdAt) || right.id.localeCompare(left.id),
            );
          const totals = calculateExpenseTotals(expenses);
          return {
            shopId: shop.id,
            businessDayId: day.id,
            loadedAt: this.#runtime.now(),
            expenses,
            ...totals,
          } satisfies ExpensesLedger;
        });
        return ledger === null
          ? err({ code: 'CONFLICT_ERROR', message: 'No active Business Day.' })
          : ok(ledger);
      } catch (cause) {
        return err(persistenceError('Could not load the local Expenses ledger.', cause));
      }
    });
  }

  async createExpense(input: ManualExpenseInput): Promise<ExpenseMutationResult> {
    return this.#mutate(async (transaction, context, now) => {
      const expense = createManualExpense(
        {
          id: this.#id<ExpenseId>(),
          shopId: context.shopId,
          businessDayId: context.businessDayId,
          createdByWorkerId: context.operator.id,
          createdAt: now,
        },
        input,
      );
      await this.#assertDayStillOpen(transaction, context);
      await transaction.expenses.put(expense);
      await transaction.audit.append(
        this.#audit(expense, context.operator, now, 'EXPENSE_CREATED', {
          description: expense.description,
          amountMinor: expense.amountMinor,
          paidFrom: expense.paidFrom,
          revision: manualExpenseLifecycle(expense).revision,
        }),
      );
      await transaction.outbox.append(
        this.#outbox(expense, now, 'EXPENSE_CREATED', {
          description: expense.description,
          amountMinor: expense.amountMinor,
          paidFrom: expense.paidFrom,
        }),
      );
      return expense;
    });
  }

  async editExpense(input: EditManualExpenseInput): Promise<ExpenseMutationResult> {
    return this.#mutate(async (transaction, context, now) => {
      const existing = await this.#currentExpense(transaction, context, input.expenseId);
      const expense = editManualExpense(existing, input, now, context.operator.id);
      await transaction.expenses.put(expense);
      const revision = manualExpenseLifecycle(expense).revision;
      await transaction.audit.append(
        this.#audit(expense, context.operator, now, 'EXPENSE_EDITED', {
          description: expense.description,
          amountMinor: expense.amountMinor,
          paidFrom: expense.paidFrom,
          revision,
        }),
      );
      await transaction.outbox.append(
        this.#outbox(expense, now, 'EXPENSE_EDITED', {
          description: expense.description,
          amountMinor: expense.amountMinor,
          paidFrom: expense.paidFrom,
        }),
      );
      return expense;
    });
  }

  async deleteExpense(expenseId: ExpenseId): Promise<ExpenseMutationResult> {
    return this.#mutate(async (transaction, context, now) => {
      const existing = await this.#currentExpense(transaction, context, expenseId);
      const expense = deleteManualExpense(existing, now, context.operator.id);
      await transaction.expenses.put(expense);
      const revision = manualExpenseLifecycle(expense).revision;
      await transaction.audit.append(
        this.#audit(expense, context.operator, now, 'EXPENSE_DELETED', {
          description: expense.description,
          amountMinor: expense.amountMinor,
          paidFrom: expense.paidFrom,
          revision,
          softDeleted: true,
        }),
      );
      await transaction.outbox.append(
        this.#outbox(expense, now, 'EXPENSE_DELETED', {
          description: expense.description,
          amountMinor: expense.amountMinor,
          paidFrom: expense.paidFrom,
          softDeleted: true,
        }),
      );
      return expense;
    });
  }

  async #mutate(
    work: (
      transaction: OperationsTransaction,
      context: MutationContext,
      now: Instant,
    ) => Promise<Expense>,
  ): Promise<ExpenseMutationResult> {
    return this.#coordinator.runExclusive(async () => {
      try {
        const contextResult = await this.#resolveMutationContext();
        if (!contextResult.ok) return contextResult;
        const context = contextResult.value;
        const now = this.#runtime.now();
        const expense = await this.#database.transaction((transaction) =>
          work(transaction, context, now),
        );
        return ok(expense);
      } catch (cause) {
        return err(mutationError(cause));
      }
    });
  }

  async #resolveMutationContext(): Promise<Result<MutationContext, ApplicationError>> {
    const shops = await this.#readModel.listActiveShops();
    const shop = shops.length === 1 ? shops[0] : undefined;
    if (shop === undefined) {
      return err({
        code: 'CONFLICT_ERROR',
        message: 'This device is not assigned to exactly one active shop.',
      });
    }
    const day = await this.#database.transaction((transaction) =>
      transaction.businessDays.getOpenForShop(shop.id),
    );
    if (day === null || day.status !== 'OPEN') {
      return err({ code: 'CONFLICT_ERROR', message: 'No active Business Day.' });
    }
    const session = await this.#readModel.getOpenWorkerSession(day.id);
    if (session === null || session.endedAt !== null) {
      return err({ code: 'CONFLICT_ERROR', message: 'A Current Operator is required.' });
    }
    const worker = await this.#database.transaction((transaction) =>
      transaction.workers.getById(session.workerId),
    );
    if (worker === null || !worker.active || worker.shopId !== shop.id) {
      return err({ code: 'CONFLICT_ERROR', message: 'The Current Operator is unavailable.' });
    }
    return ok({ shopId: shop.id, businessDayId: day.id, operator: worker });
  }

  async #assertDayStillOpen(
    transaction: OperationsTransaction,
    context: MutationContext,
  ): Promise<void> {
    const day = await transaction.businessDays.getOpenForShop(context.shopId);
    if (day === null || day.status !== 'OPEN' || day.id !== context.businessDayId) {
      throw new DomainInvariantError('The Business Day changed before the expense change committed.');
    }
  }

  async #currentExpense(
    transaction: OperationsTransaction,
    context: MutationContext,
    expenseId: ExpenseId,
  ): Promise<Expense> {
    await this.#assertDayStillOpen(transaction, context);
    const expense = await transaction.expenses.getById(expenseId);
    if (
      expense === null ||
      expense.shopId !== context.shopId ||
      expense.businessDayId !== context.businessDayId
    ) {
      throw new DomainInvariantError('The expense is not part of the current Business Day.');
    }
    return expense;
  }

  #audit(
    expense: Expense,
    worker: Worker,
    createdAt: Instant,
    eventType: Extract<
      AuditEvent['eventType'],
      'EXPENSE_CREATED' | 'EXPENSE_EDITED' | 'EXPENSE_DELETED'
    >,
    details: Readonly<Record<string, JsonValue>>,
  ): AuditEvent {
    return {
      id: this.#id<AuditEventId>(),
      shopId: expense.shopId,
      businessDayId: expense.businessDayId,
      aggregateType: 'EXPENSE',
      aggregateId: expense.id,
      eventType,
      workerId: worker.id,
      createdAt,
      details,
    };
  }

  #outbox(
    expense: Expense,
    createdAt: Instant,
    eventType: 'EXPENSE_CREATED' | 'EXPENSE_EDITED' | 'EXPENSE_DELETED',
    payload: Readonly<Record<string, JsonValue>>,
  ): OutboxEvent {
    if (expense.kind !== 'MANUAL') {
      throw new DomainInvariantError('Only manual expense mutations use the manual expense outbox path.');
    }
    const revision = manualExpenseLifecycle(expense).revision;
    return {
      id: this.#id<OutboxEventId>(),
      shopId: expense.shopId,
      businessDayId: expense.businessDayId,
      aggregateType: 'EXPENSE',
      aggregateId: expense.id,
      eventType,
      idempotencyKey: `expense:${expense.id}:${revision}:${eventType}`,
      payloadVersion: 1,
      payload: {
        expenseId: expense.id,
        businessDayId: expense.businessDayId,
        revision,
        ...payload,
      },
      createdAt,
      attemptCount: 0,
      nextAttemptAt: null,
      lastError: null,
      deliveredAt: null,
    };
  }

  #id<Id extends EntityId>(): Id {
    return parseEntityId<Id>(this.#runtime.createUuid());
  }
}
''')

function assertObject(value: unknown, label: string): asserts value is Record<string, unknown> {
  if (typeof value !== 'object' || value === null || Array.isArray(value)) {
    throw new TypeError(`${label} must be an object.`);
  }
}

function assertResult(value: unknown, label: string): asserts value is Record<string, unknown> {
  assertObject(value, label);
  if (typeof value['ok'] !== 'boolean') throw new TypeError(`${label}.ok must be boolean.`);
  if (!value['ok']) {
    assertObject(value['error'], `${label}.error`);
    if (typeof value['error']['code'] !== 'string' || typeof value['error']['message'] !== 'string') {
      throw new TypeError(`${label}.error is invalid.`);
    }
  }
}

export function assertExpensesLedgerResult(value: unknown): ExpensesLedgerResult {
  assertResult(value, 'Expenses ledger result');
  if (value['ok']) {
    assertObject(value['value'], 'Expenses ledger value');
    if (!Array.isArray(value['value']['expenses'])) {
      throw new TypeError('Expenses ledger value.expenses must be an array.');
    }
    if (
      typeof value['value']['totalExpensesMinor'] !== 'number' ||
      typeof value['value']['cashExpensesMinor'] !== 'number'
    ) {
      throw new TypeError('Expenses ledger totals are invalid.');
    }
  }
  return value as unknown as ExpensesLedgerResult;
}

export function assertExpenseMutationResult(value: unknown): ExpenseMutationResult {
  assertResult(value, 'Expense mutation result');
  if (value['ok']) {
    assertObject(value['value'], 'Expense mutation value');
    if (typeof value['value']['id'] !== 'string' || typeof value['value']['kind'] !== 'string') {
      throw new TypeError('Expense mutation value is invalid.');
    }
  }
  return value as unknown as ExpenseMutationResult;
}
''')
